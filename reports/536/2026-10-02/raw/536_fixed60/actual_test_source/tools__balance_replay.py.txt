--run invokes tools/test.ps1 twice, never Godot directly. --compare re-reads each

A PASS string alone, omitted rows, failed attempts, or different results fails.

ROOT = Path(__file__).resolve().parent.parent
ROLES = ("doho0", "guard_bat1", "guard_bat2", "heukrang0", "summon_bat1", "summon_bat2")
ROLE_IDS = {"doho0": (1, 0), "heukrang0": (2, 0), "summon_bat1": (3, 1),
            "summon_bat2": (3, 2), "guard_bat1": (4, 1), "guard_bat2": (4, 2)}
ATTEMPTS = [(3, 1), (5, 1)]
TAGS = ("ENV", "LAUNCH", "ATTEMPT", "TRIES", "CLOCK", "DT")
FLOAT_FIELDS = ("sec", "phase2", "hp0", "hp", "hp_min", "max_hp", "boss_hp")
RNG = re.compile(r"흑랑 RNG Lv(\d+) 판(\d+) (\w+) seed=(-?\d+) initial=(-?\d+)(?: final=(-?\d+))?$")
COMBAT = re.compile(r"흑랑 재생 Lv(\d+) 판(\d+) win=(true|false) attacks=(\d+) hits=(\d+) guards=(\d+) summoned=(\d+)$")
ANSI = re.compile(r"\x1b\[[0-?]*[ -/]*[@-~]")
ERROR = re.compile(r"SCRIPT ERROR|SHADER ERROR|Parse Error|Compile Error|Compilation failed|^\s*(USER )?ERROR:")


class ReplayError(ValueError):
    pass


def require(condition, message):
    if not condition:
        raise ReplayError(message)


def key(row):
    return row["level"], row["attempt"]


def integer(value, label, minimum=0):
    require(type(value) is int and value >= minimum, f"{label}: invalid integer {value!r}")
    return value


def number(value, label):
    require(isinstance(value, str), f"{label}: float must be an exact decimal string")
    try:
        result = float(value)
    except ValueError as exc:
        raise ReplayError(f"{label}: invalid decimal {value!r}") from exc
    require(math.isfinite(result), f"{label}: nonfinite decimal")
    return result


def fixed_delta(value, label):
    require(math.isclose(number(value, label), 1.0 / 60.0, rel_tol=1e-7, abs_tol=1e-10),
            f"{label}: observed delta is not 1/60")


def fixed_args(args):
    require(isinstance(args, list) and all(isinstance(x, str) for x in args), "CLI must be a string array")
    require(args.count("--fixed-fps") == 1, "actual CLI is missing or duplicating --fixed-fps")
    index = args.index("--fixed-fps")
    require(index + 1 < len(args) and args[index + 1] == "60", "actual CLI fixed FPS is not 60")
    require("--headless" in args, "fixed replay must be headless")


def scan_errors(text):
    for line in ANSI.sub("", text).splitlines():
        if ERROR.search(line):
            # Same sole exit-time allowlist as the existing e2e runner; preserve raw.
            if re.match(r"^\s*ERROR: \d+ resources still in use at exit", line):
                continue
            raise ReplayError(f"raw engine/script error: {line}")


def parse_log(raw, *, runner_output):
    try:
        text = ANSI.sub("", raw.decode("utf-8-sig", errors="strict"))
    except UnicodeDecodeError as exc:
        raise ReplayError(f"raw log is not UTF-8: {exc}") from exc
    scan_errors(text)
    require(not re.search(r"^E2E .*FAIL|^\s*✗|^실패 \d+", text, re.M), "raw run contains FAIL")
    pass_rows = re.findall(r"^E2E balance_boss PASS \(검사 (\d+)\)\s*$", text, re.M)
    require(len(pass_rows) == 1 and int(pass_rows[0]) >= 30, "missing/duplicate/incomplete balance PASS summary")
    require(len(re.findall(r"^E2E SUMMARY: 1/1 PASS\s*$", text, re.M)) == 1, "missing focused 1/1 PASS summary")
    if runner_output:
        require("E2E_SELFTEST PASS (판정 일치 5/5)" in text, "runner selftest did not pass all five probes")
        require(len(re.findall(r"^전부 PASS\s*$", text, re.M)) == 1, "test.ps1 final PASS is missing")

    tagged = {tag: [] for tag in TAGS}
    initials, finals, combat = [], [], []
    for original in text.splitlines():
        line = original.strip()
        if line.startswith("· "):
            line = line[2:]
        if line.startswith("BALANCE_REPLAY_"):
            prefix, separator, payload = line.partition(" ")
            tag = prefix.removeprefix("BALANCE_REPLAY_")
            require(tag in tagged and separator, f"unrecognized metadata row: {line}")
            try:
                row = json.loads(payload)
            except (ValueError, TypeError) as exc:
                raise ReplayError(f"malformed {tag} JSON: {payload}") from exc
            require(isinstance(row, dict), f"{tag} must be an object")
            tagged[tag].append(row)
        if line.startswith("흑랑 RNG "):
            match = RNG.fullmatch(line)
            require(match is not None, f"incomplete RNG row: {line}")
            level, attempt, actor, seed, initial, final = match.groups()
            row = {"level": int(level), "attempt": int(attempt), "actor": actor, "seed": seed, "initial": initial}
            if final is None:
                initials.append(row)
            else:
                row["final"] = final
                finals.append(row)
        if line.startswith("흑랑 재생 "):
            match = COMBAT.fullmatch(line)
            require(match is not None, f"incomplete combat row: {line}")
            lv, attempt, win, attacks, hits, guards, summoned = match.groups()
            combat.append({"level": int(lv), "attempt": int(attempt), "win": win == "true",
                           "attacks": int(attacks), "hits": int(hits), "guards": int(guards), "summoned": int(summoned)})

    require(len(tagged["ENV"]) == 1, "runtime environment header missing or duplicated")
    env = tagged["ENV"][0]
    require(env.get("schema") == 1 and env.get("scenario") == "balance_boss", "wrong runtime schema/scenario")
    require(env.get("fixed_fps") == 60 and env.get("physics_hz") == 60, "runtime fixed FPS/physics Hz is not 60/60")
    require(number(env.get("time_scale"), "time_scale") == 1.0, "runtime time_scale is not 1")
    integer(env.get("pid"), "runtime PID", 1)
    require(isinstance(env.get("user_dir"), str) and env["user_dir"], "runtime user_dir missing")
    require(isinstance(env.get("engine"), dict) and env["engine"].get("hash"), "runtime engine/hash missing")
    fixed_args(env.get("cli"))
    require(env.get("user_cli") == ["--e2e=balance_boss"], "actual user CLI is not the sole balance fixture")
    if runner_output:
        require(len(tagged["LAUNCH"]) == 1, "test.ps1 actual launch header missing or duplicated")
        launch = tagged["LAUNCH"][0]
        require(launch.get("fixed_fps") == 60 and launch.get("exe"), "wrong launch condition")
        fixed_args(launch.get("argv"))
        require(launch["argv"][-2:] == ["--", "--e2e=balance_boss"], "wrong launch scenario")
    else:
        require(not tagged["LAUNCH"], "unexpected PowerShell launch in Godot raw")

    expected_keys = [(lv, attempt, actor) for lv, attempt in ATTEMPTS for actor in ROLES]
    for rows, phase in ((initials, "initial"), (finals, "final")):
        require([(r["level"], r["attempt"], r["actor"]) for r in rows] == expected_keys,
                f"{phase}: expected all 12 ordered RNG rows (missing, duplicated, extra, or different attempt order)")
    for start, end in zip(initials, finals):
        require({k: v for k, v in end.items() if k != "final"} == start, "final RNG disagrees with its initial row")
        role, ordinal = ROLE_IDS[start["actor"]]
        expected_seed = 20260916 + start["level"] * 1000003 + start["attempt"] * 10007 + role * 1009 + ordinal * 101
        require(start["seed"] == str(expected_seed), "original seed formula changed")
        for name in ("initial", "final"):
            require(-(1 << 63) <= int(end[name]) < (1 << 63), f"RNG {name} is not signed 64-bit")

    attempts = tagged["ATTEMPT"]
    clocks = tagged["CLOCK"]
    require([key(r) for r in attempts] == ATTEMPTS, "expected both complete first-attempt combat rows in Lv3/Lv5 order")
    require([key(r) for r in clocks] == ATTEMPTS, "clock rows missing, duplicated, or reordered")
    require([key(r) for r in combat] == ATTEMPTS, "human combat rows missing, duplicated, or reordered")
    require(tagged["TRIES"] == [{"level": 3, "tries": ["처치"]}, {"level": 5, "tries": ["처치"]}],
            "first fixed seed bundle did not win, or attempt sequence is missing/different")
    for row, clock, old in zip(attempts, clocks, combat):
        require(row.get("win") is True, "recorded attempt did not win")
        require(all(row.get(k) == v for k, v in old.items()), "structured combat disagrees with original human row")
        attacks = integer(row.get("attacks"), "attacks", 1)
        hits = integer(row.get("hits"), "hits", 1)
        require(hits < attacks, "real hit and miss rolls are missing")
        require(row.get("guards") == 2 and row.get("summoned") == 2, "both guards/P2 actors required")
        used = integer(row.get("used"), "potions used")
        require(row.get("potions") == 3 and used <= 3, "original potion contract changed")
        floats = {name: number(row.get(name), name) for name in FLOAT_FIELDS}
        require(8 <= floats["sec"] <= 45 and 0 <= floats["phase2"] <= floats["sec"], "original time contract or P2 missing")
        require(0 < floats["hp"] <= floats["max_hp"] and 0 <= floats["hp_min"] <= floats["hp"]
                and floats["hp0"] == floats["max_hp"] and floats["boss_hp"] <= 0, "HP result incomplete/inconsistent")
        require(used >= 1 or floats["hp_min"] <= floats["hp0"] * 0.7, "original cost criterion did not pass")
        polls = integer(row.get("input_polls"), "input polls", 1)
        require(row.get("input_ticks_ok") is True, "input six-physics-tick contract failed")
        integer(clock.get("process_count"), "process dt count", 1)
        require(integer(clock.get("physics_count"), "physics dt count", 1) >= polls * 6, "physics clock observation incomplete")
        require(clock.get("physics_hz") == 60 and number(clock.get("time_scale"), "clock time_scale") == 1,
                "clock condition changed during attempt")
        for name in ("process_dt_min", "process_dt_max", "physics_dt_min", "physics_dt_max"):
            fixed_delta(clock.get(name), name)
        samples = [s for s in tagged["DT"] if key(s) == key(row)]
        count = integer(clock.get("dt_sample_count"), "dt sample count", 1)
        require(count <= 96 and [s.get("sample") for s in samples] == list(range(1, count + 1)), "dt samples missing/duplicate/reordered")
        for sample in samples:
            require(sample.get("physics_hz") == 60 and number(sample.get("time_scale"), "sample time_scale") == 1,
                    "dt sample condition changed")
            fixed_delta(sample.get("process_dt"), "sample process dt")
            physics_count = integer(sample.get("physics_count"), "sample physics count")
            if physics_count:
                fixed_delta(sample.get("physics_dt"), "sample physics dt")
            else:
                require(sample["sample"] == 1 and number(sample.get("physics_dt"), "initial physics dt") == 0,
                        "only initial pre-physics sample may have no physics delta yet")
            integer(sample.get("process_count"), "sample process count", 1)
            require(number(sample.get("game_time"), "sample GameClock") >= 0, "negative sample GameClock")
    require(all(key(s) in ATTEMPTS for s in tagged["DT"]), "extra dt sample attempt")
    return {"environment": env, "rng": finals, "combat": attempts, "tries": tagged["TRIES"],
            "clocks": clocks, "dt": tagged["DT"], "checks": int(pass_rows[0])}


def sha256(raw):
    return hashlib.sha256(raw).hexdigest()


def write_json(path, data):
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def verify_manifest(path):
    manifest = json.loads(path.read_text(encoding="utf-8"))
    require(manifest.get("schema") == 1 and manifest.get("condition") == "fixed60", "wrong manifest schema/condition")
    require(manifest.get("exit_code") == 0 and not manifest.get("interrupted"), "test.ps1 exited nonzero/interrupted")
    integer(manifest.get("test_pid"), "test.ps1 PID", 1)
    raw = {}
    for name in ("stdout", "stderr", "godot"):
        entry = manifest.get("raw", {}).get(name, {})
        require(isinstance(entry.get("file"), str) and entry["file"], f"manifest {name} raw missing")
        raw[name] = (path.parent / entry["file"]).read_bytes()
        require(sha256(raw[name]) == entry.get("sha256"), f"{name} raw SHA-256 differs (evidence changed)")
    scan_errors(raw["stderr"].decode("utf-8-sig", errors="strict"))
    stdout = parse_log(raw["stdout"], runner_output=True)
    godot = parse_log(raw["godot"], runner_output=False)
    require(stdout == godot, "PowerShell rows disagree with detailed Godot raw")
    return manifest, stdout


def compare_manifests(paths):
    errors, records = [], []
    for path in paths:
        try:
            records.append(verify_manifest(path))
        except (ReplayError, OSError, ValueError, KeyError, TypeError) as exc:
            errors.append(f"{path.name}: {exc}")
    report = {"schema": 1, "condition": "fixed60", "default_clock_verified": False,
              "manifests": [str(p.resolve()) for p in paths], "errors": errors, "equal_14": False, "pass": False}
    if len(records) == 2:
        left_meta, left = records[0]
        right_meta, right = records[1]
        if left_meta["test_pid"] == right_meta["test_pid"] or left["environment"]["pid"] == right["environment"]["pid"]:
            errors.append("runs are not distinct independent processes")
        differences = []
        for name in left:
            a, b = copy.deepcopy(left[name]), copy.deepcopy(right[name])
            if name == "environment":
                a.pop("pid"); b.pop("pid")
            if a != b:
                differences.append(name)
        if left_meta.get("command") != right_meta.get("command"):
            differences.append("actual test.ps1 invocation")
        report.update({"runs": [left, right], "rng_rows": len(left["rng"]), "combat_rows": len(left["combat"]),
                       "differences": differences, "equal_14": left["rng"] == right["rng"] and left["combat"] == right["combat"]})
        if differences:
            errors.append("different replay results/conditions: " + ", ".join(differences))
        report["pass"] = not errors
    return report


def user_log_path():
    override = ROOT / "override.cfg"
    user_name = "Godot/app_userdata/Joseon Hunters"
    if override.exists():
        match = re.search(r'^config/custom_user_dir_name="([^"]+)"', override.read_text(encoding="utf-8"), re.M)
        if match:
            user_name = match[1]
    return Path(os.environ["APPDATA"]) / Path(user_name) / "logs" / "godot.log"


def run_pair(out):
    require(os.name == "nt", "test.ps1 replay requires Windows")
    powershell = shutil.which("powershell.exe")
    require(powershell is not None, "powershell.exe not found")
    out.mkdir(parents=True, exist_ok=True)
    require(not any(out.glob("536_fixed60_*")) and not (out / "comparison.json").exists(),
            "output already contains replay evidence; refusing to overwrite or retry it")
    script = str(ROOT / "tools" / "test.ps1").replace("'", "''")
    command = [powershell, "-NoProfile", "-ExecutionPolicy", "Bypass", "-Command",
               "[Console]::OutputEncoding = [System.Text.UTF8Encoding]::new($false); "
               f"& '{script}' -E2e -Scenario balance_boss -FixedFps 60"]
    manifests = []
    for index in (1, 2):
        stem = f"536_fixed60_{index}"
        stdout_path, stderr_path = out / (stem + ".stdout.raw.log"), out / (stem + ".stderr.raw.log")
        godot_path, manifest_path = out / (stem + ".godot.raw.log"), out / (stem + ".result.json")
        print(f"BALANCE_REPLAY_RUN {index}/2 waiting through test.ps1 common lock; stdout={stdout_path.resolve()}", flush=True)
        interrupted = False
        with stdout_path.open("wb") as stdout_file, stderr_path.open("wb") as stderr_file:
            process = subprocess.Popen(command, cwd=ROOT, stdout=stdout_file, stderr=stderr_file)
            try:
                code = process.wait()
            except KeyboardInterrupt:
                interrupted = True
                subprocess.run(["taskkill.exe", "/PID", str(process.pid), "/T", "/F"], capture_output=True, check=False)
                code = process.wait()
        # Copy immediately before another attempt can overwrite this worktree's user log.
        source = user_log_path()
        copy_error = None
        try:
            shutil.copyfile(source, godot_path)
        except OSError as exc:
            copy_error = str(exc)
        raw = {}
        for name, path in (("stdout", stdout_path), ("stderr", stderr_path), ("godot", godot_path)):
            if path.exists():
                data = path.read_bytes()
                raw[name] = {"file": path.name, "sha256": sha256(data), "bytes": len(data)}
        manifest = {"schema": 1, "condition": "fixed60", "label": stem, "command": command,
                    "test_pid": process.pid, "exit_code": code, "interrupted": interrupted,
                    "completed_utc": datetime.now(timezone.utc).isoformat(), "godot_log_source": str(source),
                    "godot_copy_error": copy_error, "raw": raw}
        write_json(manifest_path, manifest)
        manifests.append(manifest_path)
        print(f"BALANCE_REPLAY_RUN {index}/2 exit={code} manifest={manifest_path.resolve()}", flush=True)
        if interrupted:
            raise ReplayError("replay interrupted; original partial evidence preserved")
    report = compare_manifests(manifests)
    write_json(out / "comparison.json", report)
    return report

            for name, content in (("stdout", body), ("stderr", stderr), ("godot", godot_body)):
                path = directory / f"run{index}.{name}.raw.log"
                path.write_bytes(content)
                raw[name] = {"file": path.name, "sha256": sha256(content)}
            path = directory / f"run{index}.json"
            write_json(path, {"schema": 1, "condition": "fixed60", "command": ["test.ps1", "-FixedFps", "60"],
                              "test_pid": pid if pid is not None else index + 100, "exit_code": exit_code, "raw": raw})
            return path
        first, second = fixture(1), fixture(2)
        check(compare_manifests([first, second])["pass"], "two independent complete manifests accepted")
        second = fixture(2, good.replace(b'"sec":"20.00000000000000000"', b'"sec":"20.00000000000000001"', 1))
        check(not compare_manifests([first, second])["pass"], "full decimal result difference rejected even same rounded display")
        second = fixture(2, good.replace(b"final=-6694804097080234621", b"final=-6694804097080234620", 1))
        check(not compare_manifests([first, second])["pass"], "one of 12 final RNG differences rejected")
        second = fixture(2, exit_code=7)
        check(not compare_manifests([first, second])["pass"], "nonzero exit rejected despite PASS output")
        second = fixture(2, pid=101)
        check(not compare_manifests([first, second])["pass"], "same process/evidence replay rejected")
        second = fixture(2, stderr=b"SCRIPT ERROR: stderr failure\n")
        check(not compare_manifests([first, second])["pass"], "stderr error rejected")
        second = fixture(2, godot_body=good.replace(b"final=-6694804097080234621", b"final=-6694804097080234620", 1))
        check(not compare_manifests([first, second])["pass"], "stdout/Godot evidence disagreement rejected")
        second = fixture(2)
        (directory / "run2.stdout.raw.log").write_bytes(good + b"tamper")
        check(not compare_manifests([first, second])["pass"], "raw byte tampering rejected by SHA-256")
        second = fixture(2)
        (directory / "run2.godot.raw.log").unlink()
        check(not compare_manifests([first, second])["pass"], "missing raw evidence rejected")
    print(f"BALANCE_REPLAY_TEST checks={checks} fails={len(failures)} {'FAIL' if failures else 'PASS'}")
    return 1 if failures else 0


def main():
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8", errors="backslashreplace")
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--selftest", action="store_true")
    mode.add_argument("--run", action="store_true")
    mode.add_argument("--compare", nargs=2, type=Path, metavar=("FIRST_MANIFEST", "SECOND_MANIFEST"))
    parser.add_argument("--out", type=Path, help="run evidence directory, or comparison report JSON path")
    args = parser.parse_args()
    try:
        if args.selftest:
            return selftest()
        if args.run:
            report = run_pair(args.out or (ROOT / "._tmp" / "536_fixed60"))
        else:
            report = compare_manifests(args.compare)
            if args.out:
                write_json(args.out, report)
        for error in report["errors"]:
            print("BALANCE_REPLAY_ERROR " + error)
        print(f"BALANCE_REPLAY condition=fixed60 default_clock_verified=false rng={report.get('rng_rows', 0)} "
              f"combat={report.get('combat_rows', 0)} equal_14={str(report['equal_14']).lower()} {'PASS' if report['pass'] else 'FAIL'}")
        return 0 if report["pass"] else 1
    except (ReplayError, OSError, ValueError, KeyError, TypeError) as exc:
        print("BALANCE_REPLAY FAIL " + str(exc))
        return 1


if __name__ == "__main__":
    raise SystemExit(main())
